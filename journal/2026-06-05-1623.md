"""Projects: real work units with history, not just one-liners in the sand."""
from __future__ import annotations
import json, os, time
from pathlib import Path

PROJECTS_DIR = Path(__file__).parent
INDEX_FILE = PROJECTS_DIR / "index.json"

def _load_index() -> dict:
    if INDEX_FILE.exists():
        return json.loads(INDEX_FILE.read_text())
    return {}

def _save_index(idx: dict) -> None:
    INDEX_FILE.write_text(json.dumps(idx, indent=2))

def list_projects() -> list[dict]:
    """Return all projects sorted by last_active desc."""
    idx = _load_index()
    return sorted(idx.values(), key=lambda p: p.get("last_active", 0), reverse=True)

def get_project(slug: str) -> dict | None:
    """Return one project by slug, or None."""
    idx = _load_index()
    return idx.get(slug)

def create_project(title: str, slug: str | None = None, description: str = "") -> dict:
    """Create a new project and return it."""
    idx = _load_index()
    if slug is None:
        slug = title.lower().replace(" ", "-").replace("/", "-")[:40]
    if slug in idx:
        return idx[slug]
    now = time.time()
    proj = {
        "slug": slug,
        "title": title,
        "description": description,
        "created": now,
        "last_active": now,
        "intent_log": [],  # each entry: {timestamp, text, done}
    }
    idx[slug] = proj
    _save_index(idx)
    return proj

def push_intent(slug: str, text: str) -> bool:
    """Append an intent entry to project slug. Returns True on success."""
    idx = _load_index()
    if slug not in idx:
        return False
    idx[slug]["intent_log"].append({
        "timestamp": time.time(),
        "text": text,
        "done": False,
    })
    idx[slug]["last_active"] = time.time()
    _save_index(idx)
    return True

def list_recent_intents(slug: str, limit: int = 5) -> list[dict]:
    """Return recent intent entries for this project."""
    proj = get_project(slug)
    if not proj:
        return []
    return proj["intent_log"][-limit:]

def touch_project(slug: str) -> None:
    """Update last_active without adding intent."""
    idx = _load_index()
    if slug in idx:
        idx[slug]["last_active"] = time.time()
        _save_index(idx)

def mark_intent_done(slug: str, intent_text: str) -> None:
    """Mark the most recent matching intent as done."""
    idx = _load_index()
    if slug not in idx:
        return
    for entry in reversed(idx[slug]["intent_log"]):
        if entry["text"] == intent_text and not entry.get("done"):
            entry["done"] = True
            idx[slug]["last_active"] = time.time()
            _save_index(idx)
            return

def get_or_ask_project() -> dict | None:
    """Interactive: list existing, offer 'new', return chosen project dict or None on quit."""
    projects = list_projects()
    print("\n--- Projects ---")
    if not projects:
        print("(no projects yet)")
    else:
        for i, p in enumerate(projects):
            age = _age_str(p["last_active"])
            print(f"  [{i+1}] {p['title']} (last active {age})")
    print("  [N] new project")
    print("  [Q] quit / skip project selection")
    choice = input("\nContinue existing (number) or start new [N/Q]? ").strip().lower()
    if choice == "q":
        return None
    if choice == "n":
        title = input("Project title: ").strip()
        if not title:
            return None
        return create_project(title)
    try:
        idx = int(choice) - 1
        if 0 <= idx < len(projects):
            touch_project(projects[idx]["slug"])
            return projects[idx]
    except ValueError:
        pass
    return None

def _age_str(ts: float) -> str:
    diff = time.time() - ts
    if diff < 3600:
        return f"{int(diff/60)}m ago"
    if diff < 86400:
        return f"{int(diff/3600)}h ago"
    return f"{int(diff/86400)}d ago"
